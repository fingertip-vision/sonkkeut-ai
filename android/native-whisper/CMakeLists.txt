cmake_minimum_required(VERSION 3.22)
project(sonkkeut_whisper_prototype LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CT2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/CTranslate2" CACHE PATH "Pinned CTranslate2 source")
set(CT2_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/build-arm64/libctranslate2.so" CACHE FILEPATH "Android CTranslate2 shared library")
option(BUILD_JNI "Build real CTranslate2 JNI adapter" ON)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/native/feature_cli.cpp")
  add_executable(feature_cli native/feature_cli.cpp native/log_mel.cpp)
  target_link_options(feature_cli PRIVATE "-Wl,-z,max-page-size=16384")
endif()
if(BUILD_JNI)
  add_library(ctranslate2 SHARED IMPORTED)
  set_target_properties(ctranslate2 PROPERTIES IMPORTED_LOCATION "${CT2_LIBRARY}")
  add_library(sonkkeut_whisper SHARED native/whisper_jni.cpp native/log_mel.cpp)
  target_include_directories(sonkkeut_whisper PRIVATE "${CT2_ROOT}/include" "${CT2_ROOT}/third_party")
  target_link_libraries(sonkkeut_whisper PRIVATE ctranslate2)
  target_link_options(sonkkeut_whisper PRIVATE "-Wl,-z,max-page-size=16384")
endif()
